CREATE TABLE LOG_AUDITORIA (
    ID_LOG INTEGER GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,

    -- Escritório ao qual o evento pertence
    ID_ESCRITORIO INTEGER NOT NULL,

    -- Usuário que realizou a ação; pode ficar nulo em ações automáticas
    ID_USUARIO INTEGER,
    NOME_USUARIO VARCHAR(100),

    -- Ex.: CRIAR, EDITAR, EXCLUIR, CONFIRMAR, RECUSAR, LOGIN, GERAR_PDF
    ACAO VARCHAR(30) NOT NULL,

    -- Ex.: CLIENTE, PROCESSO, AGENDAMENTO, PAGAMENTO
    TABELA_AFETADA VARCHAR(100) NOT NULL,
    ID_REGISTRO_AFETADO INTEGER,

    -- Quando a alteração for específica de um campo
    CAMPO VARCHAR(100),

    -- Valores ou estruturas JSON anteriores e posteriores
    VALOR_ANTIGO BLOB SUB_TYPE TEXT,
    VALOR_NOVO BLOB SUB_TYPE TEXT,

    -- Informações complementares: motivo de recusa, origem da ação etc.
    DETALHES BLOB SUB_TYPE TEXT,

    DATA_HORA TIMESTAMP DEFAULT CURRENT_TIMESTAMP NOT NULL,
    MAQUINA VARCHAR(100),
    IP_ORIGEM VARCHAR(45)
);

CREATE INDEX IDX_LOG_AUDITORIA_ESCRITORIO
    ON LOG_AUDITORIA (ID_ESCRITORIO);

CREATE INDEX IDX_LOG_AUDITORIA_USUARIO
    ON LOG_AUDITORIA (ID_USUARIO);

CREATE INDEX IDX_LOG_AUDITORIA_DATA_HORA
    ON LOG_AUDITORIA (DATA_HORA);

CREATE INDEX IDX_LOG_AUDITORIA_REGISTRO
    ON LOG_AUDITORIA (TABELA_AFETADA, ID_REGISTRO_AFETADO);
